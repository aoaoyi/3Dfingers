cmake_minimum_required(VERSION 3.16)
project(ReLeaf_Duo LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(ReLeaf_Duo main.cpp geometry.h)
if(MINGW AND CMAKE_SIZEOF_VOID_P EQUAL 8)
    target_include_directories(ReLeaf_Duo PRIVATE third_party/freeglut/include)
    target_link_libraries(ReLeaf_Duo PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/freeglut/lib/x64/libfreeglut.dll.a" opengl32 glu32)
    target_link_options(ReLeaf_Duo PRIVATE -static)
    add_custom_command(TARGET ReLeaf_Duo POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${CMAKE_CURRENT_SOURCE_DIR}/third_party/freeglut/bin/x64/libfreeglut.dll" "$<TARGET_FILE_DIR:ReLeaf_Duo>")
else()
    find_package(OpenGL REQUIRED)
    find_package(GLUT REQUIRED)
    target_link_libraries(ReLeaf_Duo PRIVATE OpenGL::GL OpenGL::GLU GLUT::GLUT)
endif()
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_compile_options(ReLeaf_Duo PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(geometry_test tests/geometry_test.cpp)
    # The checks use assert even when the application is built in Release.
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
        target_compile_options(geometry_test PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic)
    endif()
    add_test(NAME geometry COMMAND geometry_test)
endif()
